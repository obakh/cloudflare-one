# Internationalization (i18n)

Localize your Cloudflare Workers applications with locale detection and HTMLRewriter-based translation.

## Quick Setup

### 1. Install the package

```bash
pnpm add @repo/i18n --filter @repo/web
```

### 2. Define translations

```ts
// translations.ts
export const translations = {
  en: {
    "welcome": "Welcome!",
    "goodbye": "Goodbye!",
    "nav.home": "Home",
    "nav.about": "About",
    "auth.login": "Log In",
    "auth.logout": "Log Out",
  },
  es: {
    "welcome": "¡Bienvenido!",
    "goodbye": "¡Adiós!",
    "nav.home": "Inicio",
    "nav.about": "Acerca de",
    "auth.login": "Iniciar sesión",
    "auth.logout": "Cerrar sesión",
  },
  fr: {
    "welcome": "Bienvenue!",
    "goodbye": "Au revoir!",
    "nav.home": "Accueil",
    "nav.about": "À propos",
    "auth.login": "Connexion",
    "auth.logout": "Déconnexion",
  },

```

### 3. Create translator and detect locale

```ts
import { detectLocale } from "@repo/i18n/locale";
import { createTranslator } from "@repo/i18n/translator";
import { translations } from "./translations";

const translator = createTranslator({
  translations,
  defaultLocale: "en",
});

export default {
  async fetch(request: Request, env: Env) {
    // Detect user's locale
    const { locale, source, country } = detectLocale(request, {
      supportedLocales: ["en", "es", "fr"],
      defaultLocale: "en",
      cookieName: "locale",
      queryParam: "lang",
      useCountryFallback: true,
    });

    console.log(`Locale: ${locale} (from ${source}), Country: ${country}`);

    // Fetch and transform response
    const response = await fetch(request);
    return translator.transformFull(response, locale);
  },
};
```

## Locale Detection

Detection priority: cookie → query param → Accept-Language → country → default

```ts
import { detectLocale, LOCALES, COUNTRY_LOCALE_MAP } from "@repo/i18n/locale";

const result = detectLocale(request, {
  supportedLocales: ["en", "es", "fr", "de"],
  defaultLocale: "en",
  cookieName: "locale",      // Check cookie first
  queryParam: "lang",        // Then query param (?lang=es)
  useCountryFallback: true,  // Use cf.country as fallback
});

// Result
{
  locale: "es",
  source: "header",    // "cookie" | "query" | "header" | "country" | "default"
  country: "MX",       // From request.cf.country
  timezone: "America/Mexico_City"
}
```

### Parse Accept-Language Header

```ts
import { parseAcceptLanguage, matchLocale } from "@repo/i18n/locale";

const parsed = parseAcceptLanguage("en-US,en;q=0.9,es;q=0.8");
// [
//   { locale: "en-US", language: "en", region: "US", quality: 1 },
//   { locale: "en", language: "en", quality: 0.9 },
//   { locale: "es", language: "es", quality: 0.8 }
// ]

const match = matchLocale(parsed, ["en", "es", "fr"]);
// "en"
```

### Set Locale Cookie

```ts
import { createLocaleCookie } from "@repo/i18n/locale";

// User changes language preference
const cookie = createLocaleCookie("es", {
  maxAge: 365 * 24 * 60 * 60, // 1 year
  path: "/",
  sameSite: "Lax",
  secure: true,
});

return new Response("OK", {
  headers: { "Set-Cookie": cookie },
});
```

## HTMLRewriter Translation

Transform HTML responses using data attributes.

### HTML Template

```html
<!DOCTYPE html>
<html>
<head>
  <title data-i18n="page.title"></title>
</head>
<body>
  <h1 data-i18n="welcome"></h1>
  <nav>
    <a href="/" data-i18n="nav.home"></a>
    <a href="/about" data-i18n="nav.about"></a>
  </nav>
  
  <input 
    type="search" 
    data-i18n-placeholder="search.placeholder"
    data-i18n-title="search.title"
  />
  
  <img 
    src="/hero.jpg" 
    data-i18n-alt="hero.alt"
  />
</body>
</html>
```

### Transform Response

```ts
import { createTranslator } from "@repo/i18n/translator";

const translator = createTranslator({
  translations: {
    en: {
      "page.title": "My App",
      "welcome": "Welcome!",
      "nav.home": "Home",
      "nav.about": "About",
      "search.placeholder": "Search...",
      "search.title": "Search the site",
      "hero.alt": "Hero image",
    },
    es: {
      "page.title": "Mi App",
      "welcome": "¡Bienvenido!",
      "nav.home": "Inicio",
      "nav.about": "Acerca de",
      "search.placeholder": "Buscar...",
      "search.title": "Buscar en el sitio",
      "hero.alt": "Imagen principal",
    },
  },
  defaultLocale: "en",
});

// Transform HTML
const response = await fetch(request);
const translated = translator.transformFull(response, "es");
```

## String Interpolation

```ts
import { interpolate, pluralize } from "@repo/i18n/translator";

// Variable interpolation
const greeting = interpolate("Hello, {name}!", { name: "World" });
// "Hello, World!"

// Pluralization
const messages = pluralize(count, {
  zero: "No messages",
  one: "1 message",
  other: "{count} messages",
});
// count=0: "No messages"
// count=1: "1 message"
// count=5: "5 messages"
```

## Translation Management

### Find Missing Translations

```ts
import { findMissingTranslations } from "@repo/i18n/translator";

const english = {
  welcome: "Welcome",
  goodbye: "Goodbye",
  help: "Help",
};

const spanish = {
  welcome: "Bienvenido",
  // missing: goodbye, help
};

const missing = findMissingTranslations(english, spanish);
// ["goodbye", "help"]
```

### Merge Translations

```ts
import { mergeTranslations } from "@repo/i18n/translator";

const base = { welcome: "Welcome", goodbye: "Goodbye" };
const overrides = { welcome: "Hello" };

const merged = mergeTranslations(base, overrides);
// { welcome: "Hello", goodbye: "Goodbye" }
```

## Full Example

```ts
import { detectLocale, createLocaleCookie } from "@repo/i18n/locale";
import { createTranslator } from "@repo/i18n/translator";

const translations = {
  en: { /* ... */ },
  es: { /* ... */ },
  fr: { /* ... */ },
};

const translator = createTranslator({
  translations,
  defaultLocale: "en",
});

const SUPPORTED_LOCALES = translator.getSupportedLocales();

export default {
  async fetch(request: Request, env: Env) {
    const url = new URL(request.url);

    // Handle locale change
    if (url.pathname === "/set-locale") {
      const newLocale = url.searchParams.get("locale");
      if (newLocale && translator.isSupported(newLocale)) {
        return new Response(null, {
          status: 302,
          headers: {
            Location: "/",
            "Set-Cookie": createLocaleCookie(newLocale),
          },
        });
      }
    }

    // Detect locale
    const { locale, country, timezone } = detectLocale(request, {
      supportedLocales: SUPPORTED_LOCALES,
      defaultLocale: "en",
    });

    // Fetch origin response
    const response = await fetch(request);

    // Skip non-HTML responses
    const contentType = response.headers.get("Content-Type") || "";
    if (!contentType.includes("text/html")) {
      return response;
    }

    // Transform and return
    return translator.transformFull(response, locale);
  },
};
```

## React Integration

For client-side React apps, use `@repo/i18n/react` which wraps react-i18next.

### Setup

```tsx
// src/i18n.ts
import { initI18n } from "@repo/i18n/react";

export const i18n = initI18n({
  resources: {
    en: {
      translation: {
        welcome: "Welcome!",
        greeting: "Hello, {{name}}!",
        "nav.home": "Home",
        "nav.about": "About",
      },
    },
    es: {
      translation: {
        welcome: "¡Bienvenido!",
        greeting: "¡Hola, {{name}}!",
        "nav.home": "Inicio",
        "nav.about": "Acerca de",
      },
    },
  },
  defaultLocale: "en",
  supportedLocales: ["en", "es", "fr"],
  detectBrowserLanguage: true,
});
```

### Using Translations

```tsx
import { useTranslation, useLocale, Trans } from "@repo/i18n/react";

function MyComponent() {
  const { t } = useTranslation();
  const { locale, setLocale, supportedLocales } = useLocale();

  return (
    <div>
      <h1>{t("welcome")}</h1>
      <p>{t("greeting", { name: "World" })}</p>
      
      {/* Language switcher */}
      <select value={locale} onChange={(e) => setLocale(e.target.value)}>
        {supportedLocales.map((l) => (
          <option key={l} value={l}>{l}</option>
        ))}
      </select>
    </div>
  );
}
```

### Trans Component for Rich Text

```tsx
import { Trans } from "@repo/i18n/react";

// Translation: "Read our <link>terms of service</link>"
function Terms() {
  return (
    <Trans
      i18nKey="terms"
      components={{
        link: <a href="/terms" />,
      }}
    />
  );
}

// Translation: "Hello <bold>{{name}}</bold>!"
function Greeting({ name }: { name: string }) {
  return (
    <Trans
      i18nKey="greeting"
      values={{ name }}
      components={{
        bold: <strong />,
      }}
    />
  );
}
```

## Best Practices

1. **Keep translation keys organized** - Use namespaces like `nav.`, `auth.`, `errors.`
2. **Use interpolation** - Avoid concatenating strings, use `{variable}` placeholders
3. **Handle pluralization** - Different languages have different plural rules
4. **Set the lang attribute** - Use `transformFull()` to set `<html lang="...">`
5. **Cache translations** - Load translations once, not per request
6. **Validate translations** - Use `findMissingTranslations()` in CI/CD
7. **Use both layers** - Server-side for SEO/static content, client-side for dynamic UI
